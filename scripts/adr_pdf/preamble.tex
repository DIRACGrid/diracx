\usepackage{newunicodechar}

% Latin Modern has no arrow glyphs, and a missing character is dropped silently
% rather than reported. The ADRs use these three in prose only, never in code.
\newunicodechar{→}{\ensuremath{\rightarrow}}
\newunicodechar{↔}{\ensuremath{\leftrightarrow}}
\newunicodechar{×}{\ensuremath{\times}}

% TeX never hyphenates the first word of a paragraph, which in these ADRs'
% narrow table columns leaves long terms (\textbf{Transformation}) hanging over
% the column rule. Starting each ragged cell with zero-width glue re-enables it.
\makeatletter
\let\adr@raggedright\raggedright
\def\raggedright{\adr@raggedright\hspace{0pt}}
\makeatother

% The vocabulary and state tables are wide; give them a smaller face and tighter
% gutters so the columns keep their words.
\newcommand{\adrtablesetup}{\footnotesize\setlength{\tabcolsep}{4pt}}
\AtBeginEnvironment{longtable}{\adrtablesetup}
\AtBeginEnvironment{tabular}{\adrtablesetup}

% Long SQL and CWL lines wrap rather than run into the margin; fvextra's
% default hook arrow and indent keep a continuation from reading as a new line.
\AtBeginEnvironment{Shaded}{\small}
\fvset{breaklines=true,breakanywhere=true,breakautoindent=true,breakindent=2em}
\setlength{\emergencystretch}{3em}

\makeatletter
% A single ADR is an article and carries its own title, so the running head can
% repeat it. A combined document is a report, where the head should instead name
% the ADR the page belongs to.
\@ifundefined{chapter}
  {\newcommand{\adrrunninghead}{\@title}}
  {\renewcommand{\chaptermark}[1]{\markboth{#1}{}}%
   \newcommand{\adrrunninghead}{\nouppercase{\leftmark}}}
\AtBeginDocument{%
  \pagestyle{fancy}%
  \fancyhf{}%
  \fancyhead[L]{\footnotesize\itshape\adrrunninghead}%
  \fancyhead[R]{\footnotesize\thepage}%
  \renewcommand{\headrulewidth}{0.4pt}%
  \fancypagestyle{plain}{\fancyhf{}\renewcommand{\headrulewidth}{0pt}\fancyfoot[C]{\footnotesize\thepage}}%
}
\makeatother
